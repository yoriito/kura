48Result<void> GrpcProducerImpl::Connect(std::string target) {
49 grpc::ChannelArguments args;
50 args.SetInt(GRPC_ARG_KEEPALIVE_TIME_MS, 30000);
51 args.SetInt(GRPC_ARG_KEEPALIVE_TIMEOUT_MS, 10000);
52 args.SetInt(GRPC_ARG_KEEPALIVE_PERMIT_WITHOUT_CALLS, 0);
53 std::vector<
54 std::unique_ptr<grpc::experimental::ClientInterceptorFactoryInterface>>
55 interceptor_creators;
56 interceptor_creators.push_back(std::make_unique<GrpcInterceptorFactory>(
57 yoriito::databroker::DataBrokerType::kProducer));
58
59 auto channel = grpc::experimental::CreateCustomChannelWithInterceptors(
60 target, grpc::InsecureChannelCredentials(), args,
61 std::move(interceptor_creators));
62 stub_ = VISS::NewStub(channel);
63 return Result<void>::Ok();
64}

131Result<void> GrpcProducerImpl::UnSubscribe(
132 std::unique_ptr<ProducerSubscribeContext> subscription_id) {
133 auto raw_ptr = subscription_id.get();
134
135 std::unique_ptr<yoriito::databroker::producer::VissGrpcUnSubscribeImpl> impl =
136 std::make_unique<yoriito::databroker::producer::VissGrpcUnSubscribeImpl>(
137 stub_.get());
138
139 auto context = ClientContextFactory::Create(ClientContextFactory::Config{});
140 yoriito::viss::v1::UnsubscribeRequest request;
141 yoriito::viss::v1::UnsubscribeResponse response;
142
143 auto viss_context =
144 dynamic_cast<yoriito::databroker::producer::VissGrpcSubscribeContext*>(
145 raw_ptr);
146 if (viss_context) {
147 viss_context->Cancel();
148 impl->CreateUnsubscribeRequest(*viss_context, request);
149
150 auto result = impl->Execute(*context.get(), request, response);
151
152 return impl->ConvertUnsubscribeResponse(result, response);
153 } else {
154 std::string error_message = "subscription id is invalid";

156 return Result<void>::Err(
157 VissCommonError::CreateIllegalError(error_message));
