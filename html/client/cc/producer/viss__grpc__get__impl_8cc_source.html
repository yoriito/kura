40void VissGrpcGetImpl::CreateGetRequest(
41 yoriito::viss::v1::producer::GetTargetRequest& request,
42 const ProducerGetRequest& data) {
43 request.set_path(data.GetPath());
44 auto get_filter = request.mutable_filter();
45 auto path_filter = get_filter->mutable_paths_parameter();
46 if (data.GetFilter().has_value()) {
47 auto paths = data.GetFilter().value().GetPaths();
48 for (const std::string& path : paths) {
49 path_filter->add_paths(path);
50 }
51 }
52 return;
53}

63Result<ProducerGetResponse> VissGrpcGetImpl::ConvertGetResponse(
64 grpc::Status status,
65 yoriito::viss::v1::producer::GetTargetResponse& response,
66 const ProducerGetRequest& data) {
67 std::vector<yoriito::databroker::ProducerGetDataPoint> datapoints;
68 if (status.ok()) {
69 // VISS Error
70 if (response.has_error()) {
71 auto error = response.error().description();
72 auto code = response.error().number();
73 auto message = error + " Code:" + std::to_string(code);
74 VISS_LOG_ERROR("PRDR_GET", message);
75 return Result<ProducerGetResponse>::Err(
76 VissCommonError::CreateVissError(code, error));
77 }
78 if (!response.has_data()) {
79 // Illegal Response: No Data( actually return viss error )
80 VISS_LOG_ERROR("PRDR_GET", "Not Found " + data.GetPath());
81 return Result<ProducerGetResponse>::Err(
82 VissCommonError::CreateNotFoundError(data.GetPath()));
83 }
84
85 for (const auto& package : response.data().data()) {
86 for (const auto& dp : package.dp()) {
87 ValueType value = VissGrpcValueUtil::ConvertValueType(dp.value());
88 datapoints.push_back(
89 yoriito::databroker::ProducerGetDataPoint(package.path(), value));
90 }
91 }
92 } else {
93 auto error = status.error_message();
94 VISS_LOG_ERROR("PRDR_GET", error);
95 return Result<ProducerGetResponse>::Err(
96 VissCommonError::CreateProtocolError(error));
97 }
98 return Result<ProducerGetResponse>::Ok(
99 ProducerGetResponse(std::move(datapoints)));
100}
