45Result<void> GrpcConsumerImpl::Connect(std::string target) {
46 grpc::ChannelArguments args;
47 args.SetInt(GRPC_ARG_KEEPALIVE_TIME_MS, 30000);
48 args.SetInt(GRPC_ARG_KEEPALIVE_TIMEOUT_MS, 10000);
49 args.SetInt(GRPC_ARG_KEEPALIVE_PERMIT_WITHOUT_CALLS, 0);
50 std::vector<
51 std::unique_ptr<grpc::experimental::ClientInterceptorFactoryInterface>>
52 interceptor_creators;
53 interceptor_creators.push_back(std::make_unique<GrpcInterceptorFactory>(
54 yoriito::databroker::DataBrokerType::kConsumer));
55
56 std::shared_ptr<grpc::ChannelCredentials> credentials =
57 grpc::InsecureChannelCredentials();
58 std::shared_ptr<Channel> channel =
59 grpc::experimental::CreateCustomChannelWithInterceptors(
60 target, credentials, args, std::move(interceptor_creators));
61 stub_ = VISS::NewStub(channel);
62 return Result<void>::Ok();
63}

128Result<void> GrpcConsumerImpl::UnSubscribe(
129 std::unique_ptr<ConsumerSubscribeContext> subscription_id) {
130 auto context = ClientContextFactory::Create(ClientContextFactory::Config{});
131 yoriito::viss::v1::UnsubscribeRequest viss_request;
132 yoriito::viss::v1::UnsubscribeResponse viss_response;
133 auto raw_ptr = subscription_id.get();
134
135 std::unique_ptr<yoriito::databroker::consumer::VissGrpcUnSubscribeImpl> impl =
136 std::make_unique<yoriito::databroker::consumer::VissGrpcUnSubscribeImpl>(
137 stub_.get());
138
139 auto viss_context =
140 dynamic_cast<yoriito::databroker::consumer::VissGrpcSubscribeContext*>(
141 raw_ptr);
142 if (viss_context != nullptr) {
143 viss_context->Cancel();
144 impl->CreateUnsubscribeRequest(*viss_context, viss_request);
145
146 auto result = impl->Execute(*context.get(), viss_request, viss_response);
147
148 return impl->ConvertUnsubscribeResponse(std::move(result), viss_response);
149 } else {
150 std::string error_message = "subscription id is invalid";
151 YORIITO_LOG_ERROR("CNSR_UNSUBSCRIBE", error_message);
152 return Result<void>::Err(
153 VissCommonError::CreateIllegalError(error_message));
154 }
