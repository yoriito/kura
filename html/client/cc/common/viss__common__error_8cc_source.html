23yoriito::ErrorCode VissCommonError::ConvertVissErrorCode(int32_t code) {
24 switch (code) {
25 case yoriito::viss::v1::ERROR_UNSPECIFIED:
26 return yoriito::ErrorCode::ERROR_UNSPECIFIED;
27 case yoriito::viss::v1::ERROR_BAD_REQUEST:
28 return yoriito::ErrorCode::ERROR_BAD_REQUEST;
29 case yoriito::viss::v1::ERROR_UNAUTHORIZED:
30 return yoriito::ErrorCode::ERROR_UNAUTHORIZED;
31 case yoriito::viss::v1::ERROR_FORBIDDEN:
32 return yoriito::ErrorCode::ERROR_FORBIDDEN;
33 case yoriito::viss::v1::ERROR_NOT_FOUND:
34 return yoriito::ErrorCode::ERROR_NOT_FOUND;
35 case yoriito::viss::v1::ERROR_REQUEST_TIMEOUT:
36 return yoriito::ErrorCode::ERROR_REQUEST_TIMEOUT;
37 case yoriito::viss::v1::ERROR_TOO_MANY_REQUESTS:
38 return yoriito::ErrorCode::ERROR_TOO_MANY_REQUESTS;
39 case yoriito::viss::v1::ERROR_INTERNAL_SERVER_ERROR:
40 return yoriito::ErrorCode::ERROR_UNKNOWN;
41 case yoriito::viss::v1::ERROR_BAD_GATEWAY:
42 return yoriito::ErrorCode::ERROR_BAD_GATEWAY;
43 case yoriito::viss::v1::ERROR_SERVICE_UNAVAILABLE:
44 return yoriito::ErrorCode::ERROR_SERVICE_UNAVAILABLE;
45 case yoriito::viss::v1::ERROR_GATEWAY_TIMEOUT:
46 return yoriito::ErrorCode::ERROR_GATEWAY_TIMEOUT;
47 default:
48 return yoriito::ErrorCode::ERROR_UNKNOWN;
49 }
50}
