50Result<void> GrpcConsumerImpl::Connect(std::string target) {
51 grpc::ChannelArguments args;
52
53 args.SetInt(GRPC_ARG_KEEPALIVE_TIME_MS, 30000);
54 args.SetInt(GRPC_ARG_KEEPALIVE_TIMEOUT_MS, 10000);
55 args.SetInt(GRPC_ARG_KEEPALIVE_PERMIT_WITHOUT_CALLS, 0);
56
57 std::vector<
58 std::unique_ptr<grpc::experimental::ClientInterceptorFactoryInterface>>
59 interceptor_creators;
60 interceptor_creators.push_back(std::make_unique<GrpcInterceptorFactory>(
61 yoriito::databroker::DataBrokerType::kConsumer));
62
63 auto channel = grpc::experimental::CreateCustomChannelWithInterceptors(
64 target, grpc::InsecureChannelCredentials(), args,
65 std::move(interceptor_creators));
66 stub_ = VISS::NewStub(channel);
67
68 return Result<void>::Ok();
69}

138 yoriito::viss::v1::UnsubscribeRequest request;
139 yoriito::viss::v1::UnsubscribeResponse response;
140 auto raw_ptr = subscription_id.get();
141
142 std::unique_ptr<yoriito::databroker::consumer::VissGrpcUnSubscribeImpl> impl =
143 std::make_unique<yoriito::databroker::consumer::VissGrpcUnSubscribeImpl>(
144 stub_.get());
145
146 auto viss_context =
147 dynamic_cast<yoriito::databroker::consumer::VissGrpcSubscribeContext*>(
148 raw_ptr);
149 if (viss_context) {
150 viss_context->Cancel();
151 impl->CreateUnsubscribeRequest(*viss_context, request);
152
153 auto result = impl->Execute(*context.get(), request, response);
154
155 return impl->ConvertUnsubscribeResponse(result, response);
156 } else {
157 std::string error_message = "subscription id is invalid";
158 VISS_LOG_ERROR("CNSR_UNSUBSCRIBE", error_message);
159 return Result<void>::Err(
160 VissCommonError::CreateIllegalError(error_message));
161 }
